## Day 09 - Docker Integration with Jenkins and Github
#Ansible - Setup Docker Container for Nginx
- name Setup Docker Container for Nginx
  hosts: all
  become: yes
  tasks:
    - name: Pull Nginx image
      docker_image:
         name: nginx
         source: pull
    - name: Run Nginx container
      docker_container:
         name: nginx-container
         image: nginx
         state: started
         published_ports:
            - "8000:80"

## Teraform - EC2 with Auto Scaling Group
resource "aws_launch_configuration" "pathnex" {
    name  = "example-config"
    image_id = "ami-0abcdef12345"
    instance_type  = "t3.medium"
}

resource "aws_autoscaling_group" "pathnex" {
    desired_capacity    = 2
    max_size            = 3
    min_size            = 1
    vpc_zone_identifier = ["sunbet-123445"]
    launched__configuration = aws_launch_configuration.pathnex.id
}

## Kubernetes - Horizontal Pod Autoscaler (HPA)
apiVersion: apps/v1
kind: Deployment
metadata: 
   name: pathnex-deployment
spec:
  replicas: 1
  selector:
      matchLabels:
        app: pathnex-app
  template:
    metadata:
      labels:
        app: pathnex-app
    spec:
      containers:
        - name: nginx
          image: nginx
          ports:
            - containerPorts: 80


---
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: pathnex-hpa
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: Deployment
    name: pathnex-deployment
  minReplicas: 1
  maxReplicas: 10
  targetCPUUtilizationPercentage: 50

## Jenkinsfile - Deploy to Kubernetes
pipeline {
    agent any
    stages {
        stage('Deploy to Kubernetes') {
            steps {
                script {
                    sh 'kubectl apply -f deployment.yaml'
                }
            }
         }
    }
}
