# Ansible - Multi- Node Nginx Setup
- name: Setup Nginx on multiple nodes
  hosts: all
  become: yes
  tasks:
    - name: Install nginx
      yum:
       name: nginx
       state: present

    - name: Start nginx service
      service:
         name: nginx
         state: started
         enabled: yes

## Kubernetes - Deploy Nginx with ConfigMap and Secrets

apiVersion: v1
kind: ConfigMap
metadat:
   name: nginx-config
data:
  nginx.conf: |
    server {
      listen 80;
      server_name localhost;
      location / {
          root /usr/share/nginx/html;
          index index.html index.html
      }
   }      
--
apiVersion: v1
kind: Secret
metadata:
   name: nginx-secret
type: Opaque
data:
  password: CGJGKHBKHJK0989JJK==99

---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: pathnex-mginx
spec:
  replicas: 2
  selector:
     matchLabels:
         app: pathnex-nginx
  template:
     metadata:
        labels:
          app: pathnex-nginx
     spec:
       containers:
         - name: nginx
           image: nginx
           volumeMounts:
              - mountPath: /etc/nginx/nginx.conf
                name: nginx-config
                subPath: nginx.conf
       volumes:
         - name: nginx-config
           configMap:
              name: nginx-secret
         - name: nginx-secret
           secret:
             secretNamae: nginx-secret
## Jenkinsfile - Multi-Branch Pipeline with Docker Build

# GitLab CI/CD - Advanced Deployment with Helm
stages:
  - build
  - push
  - deploy

build:
  stage: build
  script:
      - docker build -t pathnex-web-app .
push:
  stage: push
  script:
    - docker login -u "$CI_REGISTERT_USER" -p "$CI_REGISTRY_PASSWORD"
    - docker push pathnex-web-app
  
deploy:
  stage: deploy
  script:
    - helm repo add pathnex https://charts.pathnex.com
    - helm install pathnex-nginx pathnex/nginx-ingress
